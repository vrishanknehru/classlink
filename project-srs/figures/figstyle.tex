% \tbl{name}{position}{header}{row, row, ...}
\newcommand{\tbl}[4]{%
  \node[thead] (#1) at (#2) {#3};
  \xdef\prevnode{#1}%
  \foreach \r [count=\i] in {#4}{%
    \ifodd\i\def\rowcolor{gray!32}\else\def\rowcolor{gray!12}\fi
    \node[trow, fill=\rowcolor, anchor=north] (#1-\i) at (\prevnode.south) {\r};%
    \xdef\prevnode{#1-\i}}}
